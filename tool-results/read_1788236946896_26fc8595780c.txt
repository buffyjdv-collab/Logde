   495→interface ClientInvoice {
   496→  booking: Booking;
   497→  invoiceCode: string;
   498→  items: InvoiceItem[];
   499→  subtotal: number;
   500→  tax: number;
   501→  discount: number;
   502→  total: number;
   503→  paid: number;
   504→  balance: number;
   505→  status: "paid" | "partial" | "unpaid";
   506→  createdAt: string;
   507→}
   508→
   509→function computeInvoice(booking: Booking): ClientInvoice {
   510→  const nights = Math.max(
   511→    1,
   512→    Math.round(
   513→      (new Date(booking.checkOut).getTime() -
   514→        new Date(booking.checkIn).getTime()) /
   515→        86400000
   516→    )
   517→  );
   518→  const items: InvoiceItem[] = [];
   519→  const roomCharge = booking.tariffPerDay * nights;
   520→  items.push({
   521→    label: `Room (${booking.room.roomType.name})`,
   522→    qty: nights,
   523→    rate: booking.tariffPerDay,
   524→    amount: roomCharge,
   525→  });
   526→  if (booking.extraBed) {
   527→    const eb = booking.room.roomType.extraBedPrice || 0;
   528→    items.push({
   529→      label: "Extra Bed",
   530→      qty: nights,
   531→      rate: eb,
   532→      amount: eb * nights,
   533→    });
   534→  }
   535→  const subtotal = items.reduce((s, i) => s + i.amount, 0);
   536→  const taxAmount = Math.round(subtotal * (booking.taxRate / 100) * 100) / 100;
   537→  const total = subtotal + taxAmount - booking.discount;
   538→  const paid = booking.advancePaid;
   539→  const balance = Math.max(0, total - paid);
   540→  const status: ClientInvoice["status"] =
   541→    balance <= 0 ? "paid" : paid > 0 ? "partial" : "unpaid";
   542→  const invoiceCode = `INV-${booking.bookingCode.replace(/\D/g, "").padStart(4, "0")}`;
   543→  return {
   544→    booking,
   545→    invoiceCode,
   546→    items,
   547→    subtotal,
   548→    tax: taxAmount,
   549→    discount: booking.discount,
   550→    total,
   551→    paid,
   552→    balance,
   553→    status,
   554→    createdAt: booking.checkOutActual ?? booking.checkOut,
   555→  };
   556→}
