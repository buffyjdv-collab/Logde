    68→function feeRateLabel(type: string, value: number) {
    69→  if (type === "percentage") return `${value}% of gross revenue`;
    70→  if (type === "fixed_monthly") return `${formatCurrency(value)} / month`;
    71→  if (type === "per_booking") return `${formatCurrency(value)} / booking`;
    72→  return `${value}`;

   652→function PaymentsByMethodTable({
   653→  rows,
   654→}: {
   655→  rows: RevenueReport["byPaymentMethod"];
   656→}) {
   657→  const totalAmount = rows.reduce((s, r) => s + r.amount, 0);
   658→  const totalCount = rows.reduce((s, r) => s + r.count, 0);
   659→  return (
   660→    <div className="overflow-x-auto -mx-2 px-2">
   661→      <Table>
   662→        <TableHeader>
   663→          <TableRow>
   664→            <TableHead>Method</TableHead>
   665→            <TableHead className="text-right">Count</TableHead>
   666→            <TableHead className="text-right">Amount</TableHead>
   667→            <TableHead className="text-right">% of Total</TableHead>
   668→          </TableRow>
   669→        </TableHeader>
   670→        <TableBody>
   671→          {rows.map((r) => {
   672→            const pct =
   673→              totalAmount > 0 ? (r.amount / totalAmount) * 100 : 0;
   674→            const cfg =
   675→              PAYMENT_METHOD[r.method as keyof typeof PAYMENT_METHOD];
   676→            return (
   677→              <TableRow key={r.method}>
   678→                <TableCell className="font-medium">
   679→                  <span className="inline-flex items-center gap-2">
   680→                    <span
   681→                      className={cn(
   682→                        "h-2 w-2 rounded-full",
   683→                        r.method === "cash" && "bg-emerald-500",
   684→                        r.method === "upi" && "bg-sky-500",
   685→                        r.method === "card" && "bg-violet-500",
   686→                        r.method === "bank_transfer" && "bg-amber-500",
   687→                        !cfg && "bg-zinc-500"
   688→                      )}
   689→                    />
   690→                    {cfg?.label || r.method}
   691→                  </span>
   692→                </TableCell>
   693→                <TableCell className="text-right tabular-nums">
   694→                  {r.count}
   695→                </TableCell>
   696→                <TableCell className="text-right tabular-nums">
   697→                  {formatCurrency(r.amount)}
   698→                </TableCell>
   699→                <TableCell className="text-right tabular-nums">
   700→                  {pct.toFixed(1)}%
   701→                </TableCell>
   702→              </TableRow>
   703→            );
   704→          })}
   705→          <TableRow className="border-t-2 border-border bg-muted/30">
   706→            <TableCell className="font-semibold">TOTAL</TableCell>
   707→            <TableCell className="text-right tabular-nums font-bold">
   708→              {totalCount}
   709→            </TableCell>
   710→            <TableCell className="text-right tabular-nums font-bold">
   711→              {formatCurrency(totalAmount)}
   712→            </TableCell>
   713→            <TableCell className="text-right tabular-nums font-bold">
   714→              100%
   715→            </TableCell>
   716→          </TableRow>
   717→        </TableBody>
   718→      </Table>
   719→    </div>
   720→  );
   721→}
