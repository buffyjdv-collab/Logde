    41→const addDays = (d: Date, n: number) => {
    42→  const r = new Date(d);

    47→function genCode(prefix: string, n: number) {
    48→  return `${prefix}-${String(n).padStart(4, "0")}`;
    49→}

    51→function lastMonthKey(monthsAgo: number): string {
    52→  const d = new Date();
    53→  d.setMonth(d.getMonth() - monthsAgo);
    54→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
    55→}
